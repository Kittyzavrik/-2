#include <iostream> (дает доступ к вводу и выводу информации)

int main() (функцтя начала программы)
{          (начало тела функции main)
    int n; (вводим переменную целого значения)
    std::cout << "Введите n: "; (выводиться фраза, чтобы пользователь ввел значение)
    std::cin >> n; (открывается поле для ввода с клавиатуры)

    if (n < 0) { (проверка вдург число отрицательное)
        std::cout << "n должно быть неотрицательным\n";(выводится фраза)
        return 1;(завершаем с ненулевым кодом, это значит, что есть ошибка)
    } (закрываем блок if)

    unsigned long long fact = 1; (обновляем переменную fact, здесь будет умножаться факториал i!)
    unsigned long long result = 1;(тоже самое что и с fact)

    for (int i = 1; i <= n; i++) (цикл счетчика от 1 до значения n, которое ввел пользователь)
    {
        fact = fact * i;(обновляем факториал)
        result = result * fact;(домножаем итог на полученый факториал)
    }

    std::cout << "F(" << n << ") = " << result << std::endl; ( выводится F(n) = <число>, значение result и endl-перевод строки)

    return 0; (конец программы, если 0 - то все хорошо, если не 0 - ошибка)
}

#include <iostream> (используем библиотеку для активации ввода и вывода информации)

int main() { (функция начала программы)
    int A = 0; (переменная А - целое число, имеющее значение 0)
    int B = 2 - A; (переменная В обозначает 2 - число А)

    std::cout << "A = " << A << ", B = " << B << std::endl;(выводится значение А и значение В на экран, endl - перенос на другую строку)

    return 0; (завершение кода)
}

#include <iostream> (библиотека для ввода и вывода информации)
using namespace std; (небольшое сокращение, чтобы каждый раз не писать std)

int main() { (начало кода)
    const int n = 5, m = 4; (обновляем две переменные, которые дальше нельзя будет изменять, они постоянные)

    int A[n] = {1, 3, 5, 7, 9}; (создаем массив А, из n элементов)
    int B[m] = {2, 4, 6, 8}; (создаем массив В, из n элементов)

    int C[n + m]; (массив из длин А и В)
    int i = 0, j = 0; (индексы следящие за позицией в массиве А и В)

    for (int k = 0; k < n + m; k++) { (цикл, k-индекс позиции в массиве С, заполняем по одной ячейке слева напрво)
        if (i == n) { (проверка не закончился ли массив А)
            // A закончился — кидаем остаток из B
            C[k] = B[j];( переносим значение В в массив)
            j++; (прибавляем 1)
        } else if (j == m) { (это условие, если оба массива еще не закончились)
            // B закончился — кидаем остаток из A
            C[k] = A[i];
            i++;
        } else if (A[i] <= B[j]) {
            C[k] = A[i];
            i++;
        } else {
            C[k] = B[j];
            j++;
        }
    }

    for (int k = 0; k < n + m; k++) { (печатаем все элементы С)
        cout << C[k] << " ";
    }
    cout << endl; (перевод на новую строку)

    return 0; (завершение кода)
}
